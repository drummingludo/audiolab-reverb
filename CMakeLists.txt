cmake_minimum_required(VERSION 3.22)

# Mac : un seul binaire pour Apple Silicon et Intel
set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "")
set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "")

project(AudioLabReverb VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# JUCE 8 (licence AGPLv3 : le code de ce plugin est public)
set(ALR_JUCE_PATH "" CACHE PATH "Dossier JUCE local (sinon téléchargé)")
if(ALR_JUCE_PATH)
    add_subdirectory(${ALR_JUCE_PATH} JUCE)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE
        GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
        GIT_TAG 8.0.15
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()

set(ALR_FORMATS VST3)
if(APPLE)
    list(APPEND ALR_FORMATS AU)
endif()
# AAX (Pro Tools) : SDK Avid fourni avec JUCE (licence GPL v3, compatible AGPLv3).
# Non signé, il ne se charge que dans Pro Tools Developer ; signature PACE à faire avant diffusion.
option(ALR_AAX "Fabriquer le format AAX (Mac et Windows)" ON)
if(ALR_AAX AND (APPLE OR WIN32))
    list(APPEND ALR_FORMATS AAX)
endif()
if(ALR_STANDALONE)
    list(APPEND ALR_FORMATS Standalone)
endif()

juce_add_plugin(AudioLabReverb
    PRODUCT_NAME "Audio Lab Reverb"
    PLUGIN_NAME "Lab Verb"
    DESCRIPTION "Audio Lab reverb (Chamber, Plate, Spring, Room, Hall, Church)"
    COMPANY_NAME "Transversal Studio"
    COMPANY_WEBSITE "https://transversalstudio.fr"
    COMPANY_EMAIL "ludo@transversalstudio.fr"
    BUNDLE_ID "fr.transversalstudio.audiolab.reverb"
    PLUGIN_MANUFACTURER_CODE Trsv
    PLUGIN_CODE AlRv
    FORMATS ${ALR_FORMATS}
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    VST3_CATEGORIES Fx Reverb
    AU_MAIN_TYPE kAudioUnitType_Effect
    AAX_CATEGORY AAX_ePlugInCategory_Reverb
    COPY_PLUGIN_AFTER_BUILD FALSE)

juce_add_binary_data(ALRData SOURCES
    Resources/Poppins-Regular.ttf
    Resources/Poppins-Medium.ttf
    Resources/Poppins-SemiBold.ttf
    Resources/Poppins-Bold.ttf)

target_sources(AudioLabReverb PRIVATE
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp)

target_include_directories(AudioLabReverb PRIVATE Source)
if(MSVC)
    target_compile_options(AudioLabReverb PRIVATE /utf-8)
else()
    # même arithmétique que le prototype JavaScript (pas de fusion multiplication-addition)
    target_compile_options(AudioLabReverb PRIVATE -ffp-contract=off)
endif()

target_compile_definitions(AudioLabReverb PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_REPORT_APP_USAGE=0
    JUCE_MODAL_LOOPS_PERMITTED=0)

target_link_libraries(AudioLabReverb
    PRIVATE
        ALRData
        juce::juce_audio_utils
        juce::juce_audio_processors
        juce::juce_gui_basics
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags)

# Outil de test (captures de l'interface hors DAW) — non livré
option(ALR_TESTS "Construire les outils de test" OFF)
if(ALR_TESTS)
    juce_add_console_app(ALRRenderUI PRODUCT_NAME "ALRRenderUI")
    target_sources(ALRRenderUI PRIVATE tests/render_ui.cpp)
    target_include_directories(ALRRenderUI PRIVATE Source)
    target_compile_definitions(ALRRenderUI PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    target_link_libraries(ALRRenderUI PRIVATE AudioLabReverb ALRData juce::juce_audio_utils juce::juce_gui_basics juce::juce_recommended_config_flags)

    juce_add_console_app(ALRProcessorTest PRODUCT_NAME "ALRProcessorTest")
    target_sources(ALRProcessorTest PRIVATE tests/processor_test.cpp)
    target_include_directories(ALRProcessorTest PRIVATE Source)
    target_compile_definitions(ALRProcessorTest PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    target_link_libraries(ALRProcessorTest PRIVATE AudioLabReverb ALRData juce::juce_audio_utils juce::juce_gui_basics juce::juce_recommended_config_flags)
endif()
